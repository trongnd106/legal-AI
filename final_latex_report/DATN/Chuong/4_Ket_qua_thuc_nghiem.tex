\section{Các tham số và tiêu chí đánh giá}
\label{section:4.1}

\subsection{Tiêu chí đánh giá chất lượng câu trả lời}
\label{subsection:4.1.1}
\begin{itemize}[nosep, leftmargin=*]
    \item \textbf{Keyword Accuracy:} tất cả \texttt{expected\_keywords} xuất hiện trong câu trả lời
    \item \textbf{Citation Accuracy:} \texttt{cited\_articles} khớp \texttt{expected\_citations}
    \item Phù hợp domain pháp luật: đáp án có số liệu cụ thể và trích dẫn Điều/Khoản
    \item (Tùy chọn) RAGAS: Faithfulness, Answer Relevancy
\end{itemize}

\subsection{Tiêu chí đánh giá chất lượng truy xuất ngữ cảnh}
\label{subsection:4.1.2}
\begin{itemize}[nosep, leftmargin=*]
    \item Context Recall / Context Precision (RAGAS)
    \item Đánh giá thủ công: ngữ cảnh retrieved có chứa Điều cần thiết cho multi-hop không
\end{itemize}

\subsection{Tiêu chí hiệu quả indexing và chi phí}
\label{subsection:4.1.3}
\begin{itemize}[nosep, leftmargin=*]
    \item Thời gian indexing toàn bộ pipeline
    \item Số LLM calls, token tiêu thụ ($\sim$4000--5000 calls, $\sim$10--12M tokens)
    \item Số entities L2, relationships, communities
    \item Latency query: Local vs Global; cache hit rate khi re-index
\end{itemize}

\section{Phương pháp thí nghiệm}
\label{section:4.2}

\subsection{Dữ liệu thí nghiệm}
\label{subsection:4.2.1}
\begin{itemize}[nosep, leftmargin=*]
    \item Corpus: 7 văn bản, 581 Điều --- index tại \texttt{data/labor-law/output/}
    \item Bộ câu hỏi: \texttt{TEST\_CASES} trong \texttt{tests/evaluation\_suite.py}
    \item Phân loại: \texttt{single\_hop}, \texttt{multi\_hop}, \texttt{cross\_domain}; domain \texttt{lao\_dong} ($\sim$13 cases)
    \item Cấu trúc: \texttt{question}, \texttt{expected\_keywords}, \texttt{expected\_citations}, \texttt{reasoning\_chain}
\end{itemize}

\subsection{Quy trình thí nghiệm end-to-end}
\label{subsection:4.2.2}
\begin{itemize}[nosep, leftmargin=*]
    \item Giai đoạn 1 --- Chuẩn bị: \texttt{01\_prepare\_data.py}, \texttt{02\_merge\_structural\_graph.py}
    \item Giai đoạn 2 --- Indexing: \texttt{graphrag index --root data/labor-law}
    \item Giai đoạn 3 --- Query \& Evaluate: gọi \texttt{answer\_fn} $\rightarrow$ tính keyword/citation accuracy
    \item Giai đoạn 4 --- Phân tích: so sánh mode, failure cases, case study
\end{itemize}

\subsection{Môi trường và cấu hình}
\label{subsection:4.2.3}
\begin{itemize}[nosep, leftmargin=*]
    \item Python 3.11+, \texttt{uv}, Node 20, GraphRAG CLI
    \item LLM: OpenRouter \texttt{qwen/qwen3-235b-a22b-2507}
    \item Embedding: Ollama \texttt{qwen3-embedding:4b}, dim 2560
    \item Biến môi trường: \texttt{OPENROUTER\_API\_KEY}; cache tại \texttt{data/labor-law/cache/}
\end{itemize}

\section{Các phương pháp so sánh (Baselines)}
\label{section:4.3}

\subsection{Các phương pháp so sánh}
\label{subsection:4.3.1}
\begin{itemize}[nosep, leftmargin=*]
    \item Vector-only (Naive RAG): truy xuất vector thuần
    \item Local Search: entity-centric + text units
    \item Global Search: map-reduce trên community reports
    \item Local + Multi-hop: Local Search + reasoning chain
\end{itemize}

\subsection{Kịch bản thí nghiệm}
\label{subsection:4.3.2}
\begin{itemize}[nosep, leftmargin=*]
    \item E1: Local vs Global trên toàn bộ \texttt{lao\_dong} cases
    \item E2: Vector-only vs Local Search --- single\_hop
    \item E3: Local vs Local+Multi-hop --- multi\_hop (LD010--LD012)
    \item E4: Temporal filter --- lọc văn bản theo hiệu lực
    \item E5: Contract analysis --- HĐLĐ mẫu, đo số VR triggered
    \item E6: Phân tích theo difficulty (easy/medium/hard)
\end{itemize}

\section{Kết quả thí nghiệm và phân tích}
\label{section:4.4}

\subsection{Kết quả indexing}
\label{subsection:4.4.1}
\begin{itemize}[nosep, leftmargin=*]
    \item Thống kê entities L1/L2, relationships, communities (điền sau khi index)
    \item Thời gian indexing, số LLM calls, token tiêu thụ
    \item Phân tích chất lượng extraction; cache hit rate
\end{itemize}

\subsection{So sánh hiệu quả tổng thể giữa các phương pháp}
\label{subsection:4.4.2}
\begin{itemize}[nosep, leftmargin=*]
    \item Bảng keyword\_accuracy, citation\_accuracy theo mode
    \item Kỳ vọng: Vector-only yếu multi-hop; Local mạnh single\_hop; Local+Multi-hop cải thiện LD010--LD012
\end{itemize}

\subsection{Phân tích theo category và difficulty}
\label{subsection:4.4.3}
\begin{itemize}[nosep, leftmargin=*]
    \item Accuracy theo \texttt{single\_hop} vs \texttt{multi\_hop}
    \item Accuracy theo difficulty (easy/medium/hard)
    \item Failure cases: extraction sai, retrieval miss, hallucination, temporal filter
\end{itemize}

\subsection{Case study chi tiết}
\label{subsection:4.4.4}
\begin{itemize}[nosep, leftmargin=*]
    \item Case A (LD003): Thử việc 60 ngày --- Local Search, Điều 27
    \item Case B (LD012): Lương tối thiểu $\rightarrow$ phạt NĐ 12/2022 --- Multi-hop
    \item Case C (LD011): Đơn phương chấm dứt + trợ cấp --- Điều 35 + Điều 46
    \item Case D: HĐLĐ mẫu --- VR005 phạt tiền NLĐ, Điều 127
\end{itemize}

\subsection{Phân tích hiệu năng và chi phí}
\label{subsection:4.4.5}
\begin{itemize}[nosep, leftmargin=*]
    \item Latency: Local ($<$1000ms), Global (2000--5000ms), Multi-hop
    \item Chi phí indexing: LLM calls $\times$ token/call
    \item Trade-off chất lượng vs tốc độ
\end{itemize}

\section{Thảo luận}
\label{section:4.5}

\subsection{Ý nghĩa kết quả}
\label{subsection:4.5.1}
\begin{itemize}[nosep, leftmargin=*]
    \item GraphRAG + ontology 2 lớp cải thiện multi-hop so với Vector-only
    \item Lớp 1 deterministic đảm bảo citation chính xác
    \item Community reports hỗ trợ câu hỏi tổng hợp qua Global Search
    \item Ứng dụng thực tế: tra cứu nhanh, chính xác, có trích dẫn cho NLĐ/NSDLĐ
\end{itemize}

\subsection{Hạn chế thí nghiệm}
\label{subsection:4.5.2}
\begin{itemize}[nosep, leftmargin=*]
    \item Corpus hẹp: chỉ luật lao động/BHXH
    \item Test cases domain khác chưa có index riêng
    \item Metrics keyword/citation --- chưa có đánh giá chuyên gia luật sư
    \item Phụ thuộc chất lượng LLM extraction; evaluation\_suite chưa đủ đa dạng
\end{itemize}

\subsection{So sánh với báo cáo tham chiếu GraphRAG tin tức}
\label{subsection:4.5.3}
\begin{itemize}[nosep, leftmargin=*]
    \item Tương đồng: hybrid retrieval, community reports, custom prompt, multi-hop
    \item Khác biệt: Lớp 1 deterministic; temporal = hiệu lực pháp luật; rule validator + contract analysis; benchmark citation-centric
\end{itemize}
