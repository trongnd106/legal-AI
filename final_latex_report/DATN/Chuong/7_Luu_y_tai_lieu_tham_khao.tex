\section{Lưu ý quan trọng}
\label{section:7.1}

Chương này trình bày các \textbf{lưu ý} và \textbf{khuyến nghị} thực tiễn khi triển khai, sử dụng và bảo trì hệ thống. Nội dung bổ sung cho phần kết luận (Chương~\ref{chapter:conclusion}) --- tập trung vào hướng dẫn vận hành, không lặp lại phân tích hạn chế học thuật.

\subsection{Phạm vi và giới hạn pháp lý}
\label{subsection:7.1.1}

\begin{itemize}[nosep, leftmargin=*]
    \item Hệ thống là công cụ \textbf{hỗ trợ tra cứu và rà soát sơ bộ}, \textbf{không thay thế} tư vấn pháp lý chính thức từ luật sư, cơ quan nhà nước có thẩm quyền hay phán quyết của Tòa án.
    \item Corpus hiện gồm 7 văn bản pháp luật lao động đã index --- \textbf{không} bao phủ toàn bộ quy phạm lao động (bản án, hướng dẫn nội bộ doanh nghiệp, thỏa ước lao động tập thể cụ thể).
    \item Câu trả lời do \gls{LLM} sinh dựa trên ngữ cảnh retrieved --- vẫn có rủi ro sai sót dù đã có grounding, temporal filter và rule validator.
    \item Phân tích HĐLĐ chỉ mang tính \textbf{tham khảo} trước khi ký; mọi quyết định pháp lý cuối cùng thuộc về các bên và chuyên gia pháp luật.
\end{itemize}

\subsection{Lưu ý về độ chính xác và cập nhật văn bản}
\label{subsection:7.1.2}

\begin{itemize}[nosep, leftmargin=*]
    \item Luật lao động thay đổi thường xuyên (Nghị định lương tối thiểu vùng, xử phạt hành chính\ldots). Cần \textbf{cập nhật corpus} và re-index khi có văn bản mới --- xem Mục~\ref{subsection:7.3.2}.
    \item Nếu \texttt{temporal\_warnings} xuất hiện trong câu trả lời chat, người dùng nên kiểm tra phiên bản văn bản mới nhất trên Cổng thông tin điện tử Chính phủ hoặc thuvienphapluat.vn.
    \item Số liệu cụ thể (mức lương vùng, mức phạt) có thể thay đổi theo Nghị định --- luôn đối chiếu với văn bản gốc tại thời điểm áp dụng.
    \item Chất lượng extraction L2 phụ thuộc model LLM; sau khi đổi model hoặc prompt, nên chạy lại \texttt{evaluation\_suite.py} để xác nhận citation accuracy $\geq$ 85\%.
\end{itemize}

\section{Khuyến nghị sử dụng}
\label{section:7.2}

\subsection{Đối với người lao động và người sử dụng lao động}
\label{subsection:7.2.1}

\textbf{Tra cứu pháp luật (chat):}
\begin{enumerate}[nosep, leftmargin=*]
    \item Đặt câu hỏi \textbf{cụ thể}, có ngữ cảnh (ví dụ: ``NLĐ làm việc 6 tháng được bao nhiêu ngày phép?'' thay vì ``quyền nghỉ phép'').
    \item Chọn mode phù hợp: \textbf{Local} cho tra cứu một Điều; \textbf{Global} cho câu hỏi tổng hợp (``liệt kê quyền khi nghỉ việc'').
    \item Đọc kỹ \textbf{trích dẫn} (\texttt{article\_citations}) --- mở Điều/Khoản tương ứng trong BLLĐ 2019 hoặc Nghị định liên quan để xác minh.
    \item Chú ý \textbf{cảnh báo temporal} nếu có --- có thể văn bản trích dẫn đã được thay thế.
    \item Không dùng câu trả lời làm căn cứ duy nhất cho khiếu nại, khởi kiện hay quyết định pháp lý.
\end{enumerate}

\textbf{Phân tích hợp đồng lao động:}
\begin{enumerate}[nosep, leftmargin=*]
    \item Upload file DOCX/PDF (text layer) qua giao diện web hoặc \texttt{POST /api/contract/analyze}.
    \item Chọn đúng \textbf{vùng lương} (I--IV) trong form upload --- ảnh hưởng kiểm tra VR001.
    \item Ưu tiên xử lý các mục \texttt{VIOLATION} (vi phạm) trước, sau đó \texttt{HIGH\_RISK} (rủi ro cao).
    \item Mỗi cảnh báo kèm \texttt{rule\_id} (VR008) hoặc mã LLM (L001) và căn cứ Điều --- dùng để tra cứu thêm.
    \item HĐLĐ scan/ảnh: cần OCR (\texttt{contract\_analysis/ocr\_pdf.py}, PaddleOCR) --- độ chính xác OCR ảnh hưởng kết quả phân tích.
\end{enumerate}

\subsection{Đối với cán bộ nhân sự và pháp chế doanh nghiệp}
\label{subsection:7.2.2}

\begin{itemize}[nosep, leftmargin=*]
    \item Dùng hệ thống ở giai đoạn \textbf{soạn thảo/rà soát sơ bộ} HĐLĐ mẫu, thông báo nội quy --- không thay quy trình phê duyệt nội bộ.
    \item Lưu báo cáo phân tích (\texttt{markdown\_report}) vào hồ sơ làm bằng chứng đã rà soát, kèm timestamp và phiên bản corpus.
    \item Với câu hỏi phức tạp (multi-hop, cross-domain), yêu cầu IT bật chế độ Local + Multi-hop hoặc chạy benchmark script thay vì chỉ dùng chat API mặc định.
    \item Định kỳ (6--12 tháng) kiểm tra corpus đã cập nhật NĐ lương tối thiểu và NĐ xử phạt mới nhất chưa.
\end{itemize}

\subsection{Đối với nhà phát triển và người bảo trì}
\label{subsection:7.2.3}

\textbf{Quy trình cập nhật văn bản pháp luật mới:}
\begin{enumerate}[nosep, leftmargin=*]
    \item Thêm hoặc cập nhật file \texttt{.txt} trong \texttt{data/txt/} (UTF-8, cấu trúc Điều/Khoản chuẩn).
    \item Chạy chuẩn bị L1: \texttt{python scripts/01\_prepare\_data.py --verify}.
    \item Cập nhật \texttt{data/labor-law/metadata.json} (ngày hiệu lực, \texttt{tinh\_trang}, \texttt{huong\_dan\_boi}).
    \item Index L2: \texttt{graphrag index --root data/labor-law} (hoặc \texttt{uv run poe index --root data/labor-law}).
    \item Merge graph: \texttt{python scripts/02\_merge\_structural\_graph.py --root data/labor-law}.
    \item Kiểm tra artifacts: \texttt{merged\_entities.parquet}, \texttt{merged\_relationships.parquet} trong \texttt{output/}.
    \item Restart API: \texttt{uvicorn api.main:app --host 0.0.0.0 --port 8000}.
    \item Chạy regression: \texttt{python -m tests.evaluation\_suite} (hoặc script wrapper tương đương) --- xác nhận accuracy không giảm.
\end{enumerate}

\textbf{Giám sát vận hành (monitoring):}
\begin{itemize}[nosep, leftmargin=*]
    \item \textbf{Latency:} Local Search P95 $<$ 10\,s; Global Search có thể $>$ 20\,s --- cảnh báo nếu vượt timeout API 90\,s.
    \item \textbf{Health check:} \texttt{GET /api/health} --- field \texttt{graphrag\_ready} phải \texttt{true}.
    \item \textbf{Citation quality:} nếu keyword accuracy $<$ 85\% sau cập nhật, review \texttt{prompts/extract\_graph\_simple.txt} và community report prompt.
    \item \textbf{LLM API:} theo dõi rate limit OpenRouter (503/429); bật cache JSON (\texttt{data/labor-law/cache/}).
    \item \textbf{A/B định kỳ:} so sánh Local vs Global vs Multi-hop trên \texttt{TEST\_CASES} domain \texttt{lao\_dong}.
\end{itemize}

Chi tiết cấu hình xem Phụ lục~A; cấu trúc thư mục xem Phụ lục~C.

\label{section:7.4}

\subsection{Bảo vệ dữ liệu hợp đồng lao động}
\label{subsection:7.4.1}

\begin{itemize}[nosep, leftmargin=*]
    \item HĐLĐ upload chứa thông tin cá nhân (CMND, lương, địa chỉ) --- \textbf{không} lưu trữ lâu dài trên server demo trừ khi có chính sách rõ ràng.
    \item Neo4j \texttt{ContractSession} (nếu bật): subgraph tách biệt đồ thị HĐLĐ và đồ thị \texttt{labor-law}; khuyến nghị cấu hình TTL/xóa phiên định kỳ trên production.
    \item Khuyến nghị production: mã hoá file upload at-rest; xóa file tạm sau khi phân tích; log không ghi nội dung HĐLĐ đầy đủ.
\end{itemize}

\subsection{Bảo mật API và triển khai production}
\label{subsection:7.4.2}

\begin{itemize}[nosep, leftmargin=*]
    \item \textbf{API authentication:} demo hiện chưa có JWT/API key --- cần bổ sung trước triển khai công khai.
    \item \textbf{Rate limiting:} giới hạn request/IP (khuyến nghị 60--100 req/phút) để tránh lạm dụng LLM API.
    \item \textbf{CORS:} cấu hình \texttt{allow\_origins} cụ thể thay vì \texttt{*} trong production.
    \item \textbf{HTTPS:} bắt buộc khi deploy internet; reverse proxy (Nginx) trước FastAPI và Next.js.
    \item \textbf{Secrets:} API key chỉ trong \texttt{.env} hoặc secret manager; không hardcode trong \texttt{settings.yaml} commit git.
\end{itemize}
